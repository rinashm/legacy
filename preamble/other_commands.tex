% text definition
% Выделение определения Сейчас --- полужирное начертание, но эту
% команду можно изменить на что угодно. Можно поросить её покрасить
% текст или выдать что-то совсем другое
\newcommand*{\tdef}[1]{%
	\textbf{#1}%
}

% Как отображается номер для subsection
\renewcommand*{\thesubsection}{\arabic{subsection}}

\renewcommand*{\leq}{\leqslant}
\renewcommand*{\geq}{\geqslant}

\newcommand*{\nextitem}{%
	\scalebox{1.2}{%
			\rotatebox{-90}{%
				\fbox{%
					\mbox{$\leq$}%
				}%
			}%
	}%
}
